\section{Part 2}
\begin{frame}{HardBPR：不改模型，只改损失}
\framesubtitle{零参数 $\cdot$ 零推理开销 $\cdot$ 模型无关 $\cdot$ embedding来源无关}

\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{方法}

\vspace{2mm}
\small
标准 BPR：
$$\mathcal{L}_{\text{BPR}} = -\frac{1}{|\mathcal{D}|}\sum_{(u,i,j)}\ln\sigma(\hat{y}_{ui}-\hat{y}_{uj})$$

\vspace{1mm}
\textbf{HardBPR (本文)}：
$$\mathcal{L}_{\text{HB}} = -\frac{1}{|\mathcal{D}|}\sum_{(u,i,j)} \!\!\!\underbrace{(1 + \beta\!\cdot\!\text{cos}(\mathbf{s}_i,\mathbf{s}_j))}_{w_{ij}}\!\!\! \cdot \ln\sigma(\hat{y}_{ui}-\hat{y}_{uj})$$

\vspace{2mm}
\begin{itemize}
\setlength\itemsep{3pt}
\item $\mathbf{s}_i$, $\mathbf{s}_j$：冻结的 LLM/语义物品向量
\item $\beta=0.5$：语义权重；$\text{cos}$高 $\to$ 困难样本 $\to$ 更高权重
\item \textbf{不改前向传播}，与标准 LightGCN 完全相同
\item \textbf{不加参数}，LLM仅用于离线 loss 计算
\item 适用于任何 frozen 语义嵌入（RLMRec / MiniLM / ...）
\end{itemize}

\vspace{2mm}
\textbf{Information Bottleneck 解释}

\scriptsize
损失层加权 $\to$ 保持压缩结构 + 聚焦困难样本 \\
数据层硬负采样 $\to$ 破坏压缩结构 $\to$ 性能退化 (0.0397 $\to$ 0.0289)
\end{column}
